cmake_minimum_required(VERSION 3.16)
project(rphand_ros LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
# ROS loads message type support dynamically; only the SDK core is static.
set(BUILD_SHARED_LIBS ON)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/DexHandState.msg" DEPENDENCIES std_msgs)

set(RPHAND_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(RPHAND_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(RPHAND_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF)
set(CMAKE_INSTALL_LIBDIR lib CACHE PATH "" FORCE)
add_subdirectory(../.. sdk)
add_executable(rphand_node src/rphand_node.cpp)
target_compile_features(rphand_node PRIVATE cxx_std_17)
target_compile_options(rphand_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(rphand_node rphand::rphand)
ament_target_dependencies(rphand_node rclcpp std_msgs std_srvs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(rphand_node "${typesupport}")
rphand_runtime(rphand_node "$ORIGIN/..")
install(TARGETS rphand_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/command.py scripts/wave.py scripts/rphand_joy_mapper.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/wave_motion.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
