cmake_minimum_required(VERSION 3.16)
project(roboparty_dexhand_ros LANGUAGES C CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(rosidl_default_generators REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "msg/DexHandState.msg" DEPENDENCIES std_msgs)

set(ROBOPARTY_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(DH116S_BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(DH116S_BUILD_SMOKE OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_INSTALL_LIBDIR lib CACHE PATH "" FORCE)
add_subdirectory(../.. sdk)
add_executable(dexhand_node src/dexhand_node.cpp)
target_compile_features(dexhand_node PRIVATE cxx_std_17)
target_compile_options(dexhand_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(dexhand_node roboparty::dexhand_sdk)
ament_target_dependencies(dexhand_node rclcpp std_msgs std_srvs)
rosidl_get_typesupport_target(typesupport ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(dexhand_node "${typesupport}")
dh116s_runtime(dexhand_node "$ORIGIN/..")
install(TARGETS dexhand_node DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/command.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
