cmake_minimum_required(VERSION 3.16)
project(roboparty_dexhand_sdk VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "roboparty_dexhand_sdk requires Linux SocketCAN")
endif()
include(GNUInstallDirs)
string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" DH116S_PROCESSOR)
if(DH116S_PROCESSOR MATCHES "^(aarch64|arm64)$")
    set(DH116S_VENDOR_ARCH aarch64)
elseif(DH116S_PROCESSOR MATCHES "^(x86_64|amd64)$")
    set(DH116S_VENDOR_ARCH x86_64)
else()
    message(FATAL_ERROR "No bundled vendor library for ${CMAKE_SYSTEM_PROCESSOR}; supported: aarch64, x86_64")
endif()
set(DH116S_VENDOR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lhandpro")
set(DH116S_VENDOR_LIBRARY "${DH116S_VENDOR_ROOT}/${DH116S_VENDOR_ARCH}/lib/libLHandProLib.so")
# Stage the runtime next to build outputs so the build directory is relocatable.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/lib")
configure_file("${DH116S_VENDOR_LIBRARY}" "${CMAKE_CURRENT_BINARY_DIR}/lib/libLHandProLib.so" COPYONLY)
add_library(dh116s_vendor SHARED IMPORTED)
set_target_properties(dh116s_vendor PROPERTIES
    IMPORTED_LOCATION "${CMAKE_CURRENT_BINARY_DIR}/lib/libLHandProLib.so")
message(STATUS "Using bundled LHandProLib 20260921 (${DH116S_VENDOR_ARCH})")
function(dh116s_runtime target install_path)
    set_target_properties(${target} PROPERTIES
        BUILD_RPATH_USE_ORIGIN TRUE BUILD_RPATH "$ORIGIN/lib"
        INSTALL_RPATH "${install_path}")
    # Prefer the packaged ABI over older copies in LD_LIBRARY_PATH.
    target_link_options(${target} PRIVATE "-Wl,--disable-new-dtags")
endfunction()
find_package(Threads REQUIRED)
add_library(roboparty_dexhand_sdk src/hand.cpp src/socketcan_transport.cpp)
add_library(roboparty::dexhand_sdk ALIAS roboparty_dexhand_sdk)
add_library(dh116s::sdk ALIAS roboparty_dexhand_sdk)
target_compile_features(roboparty_dexhand_sdk PUBLIC cxx_std_17)
target_include_directories(roboparty_dexhand_sdk PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:include>
    PRIVATE "${DH116S_VENDOR_ROOT}/include")
target_link_libraries(roboparty_dexhand_sdk PRIVATE dh116s_vendor Threads::Threads)
dh116s_runtime(roboparty_dexhand_sdk "$ORIGIN")
target_compile_options(roboparty_dexhand_sdk PRIVATE -Wall -Wextra -Wpedantic)
option(DH116S_BUILD_DEMO "Build dual-hand wave demo" ON)
if(DH116S_BUILD_DEMO)
    add_executable(dh116s_wave_demo examples/wave_demo.cpp)
    target_link_libraries(dh116s_wave_demo PRIVATE roboparty::dexhand_sdk Threads::Threads)
    target_compile_options(dh116s_wave_demo PRIVATE -Wall -Wextra -Wpedantic)
    dh116s_runtime(dh116s_wave_demo "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
    install(TARGETS dh116s_wave_demo RUNTIME DESTINATION bin)
endif()
install(FILES "${DH116S_VENDOR_LIBRARY}" DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(TARGETS roboparty_dexhand_sdk ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
    LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
option(DH116S_BUILD_SMOKE "Build a small hardware-free API/lifecycle check" OFF)
if(DH116S_BUILD_SMOKE)
    enable_testing()
    add_executable(dh116s_basic_smoke tests/basic_smoke.cpp)
    target_link_libraries(dh116s_basic_smoke PRIVATE roboparty::dexhand_sdk)
    dh116s_runtime(dh116s_basic_smoke "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
    add_test(NAME dh116s_basic_smoke COMMAND dh116s_basic_smoke)
endif()
